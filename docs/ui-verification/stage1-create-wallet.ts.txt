import { chromium } from 'playwright-core'
const browser = await chromium.launch()
const ctx = await browser.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 2 })
const page = await ctx.newPage()
const errs: string[] = []
page.on('pageerror', (e) => errs.push(e.message))
page.on('console', (m) => { if (m.type() === 'error') errs.push(m.text().slice(0, 200)) })
await page.goto('http://localhost:5173/', { waitUntil: 'networkidle' })
await page.getByRole('button', { name: 'Connect wallet' }).first().click()
await page.screenshot({ path: '/tmp/shots/v2-chooser.png' })
await page.getByRole('button', { name: /Create a SplitPay testnet wallet/ }).click()
await page.getByLabel('Choose a 6-digit PIN').fill('123456')
await page.getByLabel('Repeat the PIN').fill('123456')
await page.getByLabel(/I understand this is a test wallet/).check()
await page.screenshot({ path: '/tmp/shots/v2-create-wallet.png' })
await page.getByRole('button', { name: 'Create wallet' }).click()
await page.getByRole('dialog', { name: 'Your testnet wallet' }).waitFor({ timeout: 30000 })
const addr = (await page.locator('code').first().innerText()).trim()
await page.screenshot({ path: '/tmp/shots/v2-wallet-manage.png' })
console.log('ADDRESS', addr)
await ctx.storageState({ path: '/tmp/e2e/embedded-state.json' })
console.log('errors', errs)
await browser.close()
